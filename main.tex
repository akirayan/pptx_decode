% --------------------
% Float パラメータ調整
% (B5 paper では大型テーブルが topfraction=0.7 を超えて
%  float キューに滞留しページ 11 等でテーブルが消える問題を解消)
% --------------------
\renewcommand{\topfraction}{0.9}       % ページ上部への float 最大占有率 0.7→0.9
\renewcommand{\textfraction}{0.1}      % テキスト最低占有率 0.2→0.1
\renewcommand{\floatpagefraction}{0.85}% float 専用ページの最低充填率
\setcounter{topnumber}{5}              % ページ上部の float 最大数 2→5
\setcounter{totalnumber}{10}           % ページあたり float 最大数 3→10
